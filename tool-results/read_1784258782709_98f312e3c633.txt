   124→function LoginScreen({ onLogin }: { onLogin: (u: AdminUser) => void }) {

   129→

   131→    e.preventDefault()

   133→    try {
   134→      const { user } = await api.login(email, password)
   135→      toast({ title: `Welcome, ${user.name}` })
   136→      onLogin(user)
   137→    } catch (e) {
   138→      toast({
   139→        title: 'Login failed',
   140→        description: e instanceof Error ? e.message : 'Unknown error',
   141→        variant: 'destructive',
   142→      })
   143→    } finally {
   144→      setLoading(false)
   145→    }
   146→  }

   148→  return (

   212→function ReportsAdmin() {

   222→      toast({ title: `Marked as ${status}` })
   223→      load()
   224→    } catch (e) {
   225→      toast({
   226→        title: 'Update failed',
   227→        description: e instanceof Error ? e.message : 'Unknown error',
   228→        variant: 'destructive',
   229→      })
   230→    }
   231→  }
   232→
   233→  if (!reports) {
   234→    return <Skeleton className="h-64 w-full" />
   235→  }
   236→
   237→  return (
   238→    <Card>
   239→      <CardHeader>
   240→        <CardTitle className="text-base">Community reports ({reports.length})</CardTitle>
   241→      </CardHeader>
   242→      <CardContent className="space-y-2">

   359→function UtilityFormDialog({
   360→  open, utility, onClose, onSaved,
   361→}: {
   362→  open: boolean
   363→  utility: Utility | null
   364→  onClose: () => void
   365→  onSaved: () => void
   366→}) {
   367→  const [form, setForm] = useState<Partial<Utility>>({})
   368→  const [saving, setSaving] = useState(false)
   369→  const { toast } = useToast()
   370→
   371→  useEffect(() => {
   372→    setForm(utility ?? {
   373→      pwsid: '', name: '', city: '', state: '', zipCodes: '', county: '',
   374→      population: 0, systemType: 'Community', sourceType: 'Surface',
   375→      treatmentStatus: 'Treated', latitude: null, longitude: null,
   376→      website: '', notes: '',
   377→    })
   378→  }, [utility, open])
   379→
   380→  const save = async () => {
   381→    if (!form.pwsid || !form.name || !form.state) {
   382→      toast({ title: 'PWSID, name, and state are required', variant: 'destructive' })
   383→      return
   384→    }
   385→    setSaving(true)
   386→    try {
   387→      if (utility) {
   388→        await api.updateUtility(utility.id, form)
   389→        toast({ title: 'Utility updated' })
   390→      } else {
   391→        await api.createUtility(form)
   392→        toast({ title: 'Utility created' })
   393→      }
   394→      onSaved()
   395→    } catch (e) {
   396→      toast({
   397→        title: 'Save failed',
   398→        description: e instanceof Error ? e.message : 'Unknown error',
   399→        variant: 'destructive',
   400→      })
   401→    } finally {
   402→      setSaving(false)
   403→    }
   404→  }
   405→
   406→  return (
   407→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   408→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   409→        <DialogHeader>
   410→          <DialogTitle>{utility ? 'Edit utility' : 'Add utility'}</DialogTitle>
   411→        </DialogHeader>
   412→        <div className="grid gap-3 py-2 sm:grid-cols-2">
   413→          <FormField label="PWSID *" value={form.pwsid ?? ''} onChange={(v) => setForm({ ...form, pwsid: v })} />
   414→          <FormField label="State *" value={form.state ?? ''} onChange={(v) => setForm({ ...form, state: v.toUpperCase().slice(0, 2) })} />
   415→          <div className="sm:col-span-2">
   416→            <FormField label="Utility name *" value={form.name ?? ''} onChange={(v) => setForm({ ...form, name: v })} />
   417→          </div>
   418→          <FormField label="City" value={form.city ?? ''} onChange={(v) => setForm({ ...form, city: v })} />
   419→          <FormField label="County" value={form.county ?? ''} onChange={(v) => setForm({ ...form, county: v })} />
   420→          <div className="sm:col-span-2">
   421→            <FormField label="ZIP codes (comma-separated)" value={form.zipCodes ?? ''} onChange={(v) => setForm({ ...form, zipCodes: v })} />
   422→          </div>
   423→          <FormField label="Population" type="number" value={String(form.population ?? 0)} onChange={(v) => setForm({ ...form, population: Number(v) })} />
   424→          <FormField label="Website" value={form.website ?? ''} onChange={(v) => setForm({ ...form, website: v })} />
   425→          <FormField label="Latitude" type="number" value={form.latitude ? String(form.latitude) : ''} onChange={(v) => setForm({ ...form, latitude: v ? Number(v) : null })} />
   426→          <FormField label="Longitude" type="number" value={form.longitude ? String(form.longitude) : ''} onChange={(v) => setForm({ ...form, longitude: v ? Number(v) : null })} />
   427→          <FormSelectField label="System type" value={form.systemType ?? 'Community'} onChange={(v) => setForm({ ...form, systemType: v })} options={['Community', 'Non-Transient Non-Community', 'Transient Non-Community']} />
   428→          <FormSelectField label="Source type" value={form.sourceType ?? 'Surface'} onChange={(v) => setForm({ ...form, sourceType: v })} options={['Surface', 'Ground', 'Mixed']} />
   429→          <FormSelectField label="Treatment status" value={form.treatmentStatus ?? 'Treated'} onChange={(v) => setForm({ ...form, treatmentStatus: v })} options={['Treated', 'Untreated', 'Mixed']} />
   430→          <div className="sm:col-span-2">
   431→            <Label className="text-xs">Notes</Label>
   432→            <Textarea
   433→              value={form.notes ?? ''}
   434→              onChange={(e) => setForm({ ...form, notes: e.target.value })}
   435→              rows={2}
   436→            />
   437→          </div>
   438→        </div>
   439→        <DialogFooter>
   440→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   441→          <Button onClick={save} disabled={saving}>
   442→            {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : null}
   443→            {utility ? 'Save changes' : 'Create utility'}
   444→          </Button>
   445→        </DialogFooter>
   446→      </DialogContent>
   447→    </Dialog>
   448→  )
   449→}

   451→function FormField({
   452→  label, value, onChange, type = 'text',
   453→}: {
   454→  label: string
   455→  value: string
   456→  onChange: (v: string) => void
   457→  type?: string
   458→}) {
   459→  return (
   460→    <div>
   461→      <Label className="text-xs">{label}</Label>
   462→      <Input type={type} value={value} onChange={(e) => onChange(e.target.value)} />
   463→    </div>
   464→  )
   465→}
   466→
   467→function FormSelectField({
   468→  label, value, onChange, options,
   469→}: {
   470→  label: string
   471→  value: string
   472→  onChange: (v: string) => void
   473→  options: string[]
   474→}) {
   475→  return (
   476→    <div>
   477→      <Label className="text-xs">{label}</Label>
   478→      <Select value={value} onValueChange={onChange}>
   479→        <SelectTrigger><SelectValue /></SelectTrigger>
   480→        <SelectContent>
   481→          {options.map((o) => <SelectItem key={o} value={o}>{o}</SelectItem>)}
   482→        </SelectContent>
   483→      </Select>
   484→    </div>
   485→  )
   486→}

   537→// ── Import / Export ────────────────────────────────────────────────────
   538→function DataAdmin() {
   539→  const [table, setTable] = useState<'utilities' | 'contaminants' | 'samples' | 'reports' | 'volunteers'>('utilities')
   540→  const [importFormat, setImportFormat] = useState<'csv' | 'json'>('json')
   541→  const [importText, setImportText] = useState('')
   542→  const [importing, setImporting] = useState(false)
   543→  const [result, setResult] = useState<{ imported: number; errors: string[] } | null>(null)
   544→  const { toast } = useToast()
   545→
   546→  const handleExport = (format: 'csv' | 'json') => {
   547→    window.location.href = api.exportUrl(format, table)
   548→    toast({ title: `Exporting ${table} as ${format.toUpperCase()}` })
   549→  }
   550→
   551→  const handleImport = async () => {
   552→    if (!importText.trim()) {
   553→      toast({ title: 'Paste data to import first', variant: 'destructive' })
   554→      return
   555→    }
   556→    setImporting(true)
   557→    setResult(null)
   558→    try {
   559→      const r = await api.importData(table, importFormat, importText)
   560→      setResult(r)
   561→      toast({
   562→        title: `Imported ${r.imported} ${table}`,
   563→        description: r.errors.length ? `${r.errors.length} errors` : 'All rows succeeded',
   564→        variant: r.errors.length ? 'destructive' : 'default',
   565→      })
   566→    } catch (e) {
   567→      toast({
   568→        title: 'Import failed',
   569→        description: e instanceof Error ? e.message : 'Unknown error',
   570→        variant: 'destructive',
   571→      })
   572→    } finally {
   573→      setImporting(false)
   574→    }
   575→  }
   576→
   577→  return (
   578→    <div className="space-y-4">
   579→      {/* Export */}
   580→      <Card>
   581→        <CardHeader>
   582→          <CardTitle className="flex items-center gap-2 text-base">
   583→            <Download className="h-4 w-4 text-primary" />
   584→            Export data
   585→          </CardTitle>
   586→          <p className="text-sm text-muted-foreground">
   587→            Download the entire database as CSV or JSON. Perfect for backups, sharing, or moving to another host.
   588→          </p>
   589→        </CardHeader>
   590→        <CardContent>
   591→          <div className="flex flex-wrap items-end gap-3">
   592→            <div>
   593→              <Label className="text-xs">Table</Label>
   594→              <Select value={table} onValueChange={(v: typeof table) => setTable(v)}>
   595→                <SelectTrigger className="w-44"><SelectValue /></SelectTrigger>
   596→                <SelectContent>
   597→                  <SelectItem value="utilities">Utilities</SelectItem>
   598→                  <SelectItem value="contaminants">Contaminants</SelectItem>
   599→                  <SelectItem value="samples">Samples</SelectItem>
   600→                  <SelectItem value="reports">Reports</SelectItem>
   601→                  <SelectItem value="volunteers">Volunteers</SelectItem>
   602→                </SelectContent>
   603→              </Select>
   604→            </div>
   605→            <Button onClick={() => handleExport('csv')} variant="outline">
   606→              <FileSpreadsheet className="h-4 w-4" />
   607→              Download CSV
   608→            </Button>
   609→            <Button onClick={() => handleExport('json')} variant="outline">
   610→              <FileJson className="h-4 w-4" />
   611→              Download JSON
   612→            </Button>
   613→          </div>
   614→        </CardContent>
   615→      </Card>
   616→
   617→      {/* Import */}
   618→      <Card>
   619→        <CardHeader>
   620→          <CardTitle className="flex items-center gap-2 text-base">
   621→            <Upload className="h-4 w-4 text-primary" />
   622→            Import data
   623→          </CardTitle>
   624→          <p className="text-sm text-muted-foreground">
   625→            Paste CSV or JSON content below. For utilities and contaminants, existing records (matched by PWSID or slug) will be updated; otherwise new records are created. For samples and reports, new rows are always appended.
   626→          </p>
   627→        </CardHeader>
   628→        <CardContent className="space-y-3">
   629→          <div className="flex flex-wrap items-end gap-3">
   630→            <div>
   631→              <Label className="text-xs">Table</Label>
   632→              <Select value={table} onValueChange={(v: typeof table) => setTable(v)}>
   633→                <SelectTrigger className="w-44"><SelectValue /></SelectTrigger>
   634→                <SelectContent>
   635→                  <SelectItem value="utilities">Utilities</SelectItem>
   636→                  <SelectItem value="contaminants">Contaminants</SelectItem>
   637→                  <SelectItem value="samples">Samples</SelectItem>
   638→                  <SelectItem value="reports">Reports</SelectItem>
   639→                  <SelectItem value="volunteers">Volunteers</SelectItem>
   640→                </SelectContent>
   641→              </Select>
   642→            </div>
   643→            <div>
   644→              <Label className="text-xs">Format</Label>
   645→              <Select value={importFormat} onValueChange={(v: 'csv' | 'json') => setImportFormat(v)}>
   646→                <SelectTrigger className="w-28"><SelectValue /></SelectTrigger>
   647→                <SelectContent>
   648→                  <SelectItem value="json">JSON</SelectItem>
   649→                  <SelectItem value="csv">CSV</SelectItem>
   650→                </SelectContent>
   651→              </Select>
   652→            </div>
   653→            <div>
   654→              <Label className="text-xs">&nbsp;</Label>
   655→              <input
   656→                type="file"
   657→                accept=".json,.csv"
   658→                onChange={async (e) => {
   659→                  const f = e.target.files?.[0]
   660→                  if (!f) return
   661→                  const text = await f.text()
   662→                  setImportText(text)
   663→                  if (f.name.endsWith('.csv')) setImportFormat('csv')
   664→                  if (f.name.endsWith('.json')) setImportFormat('json')
   665→                }}
   666→                className="block text-xs file:mr-2 file:rounded-md file:border-0 file:bg-primary file:px-3 file:py-1.5 file:text-primary-foreground hover:file:bg-primary/90"
   667→              />
   668→            </div>
   669→          </div>
   670→          <Textarea
   671→            value={importText}
   672→            onChange={(e) => setImportText(e.target.value)}
   673→            placeholder={
   674→              importFormat === 'json'
   675→                ? '[\n  {"pwsid":"IL0316040","name":"City of Chicago",...}\n]'
   676→                : 'pwsid,name,state,city\nIL0316040,City of Chicago,IL,Chicago'
   677→            }
   678→            rows={10}
   679→            className="font-mono text-xs"
   680→          />
   681→          <Button onClick={handleImport} disabled={importing}>
   682→            {importing ? (
   683→              <><Loader2 className="h-4 w-4 animate-spin" /> Importing...</>
   684→            ) : (
   685→              <><Upload className="h-4 w-4" /> Import {importFormat.toUpperCase()}</>
   686→            )}
   687→          </Button>
   688→
   689→          {result && (
   690→            <div className={cn(
   691→              'rounded-lg border p-3 text-sm',
   692→              result.errors.length
   693→                ? 'border-amber-200 bg-amber-50 text-amber-800'
   694→                : 'border-emerald-200 bg-emerald-50 text-emerald-800'
   695→            )}>
   696→              <div className="flex items-center gap-2 font-medium">
   697→                {result.errors.length ? <AlertCircle className="h-4 w-4" /> : <CheckCircle2 className="h-4 w-4" />}
   698→                Imported {result.imported} {result.errors.length === 0 ? 'successfully' : `with ${result.errors.length} errors`}
   699→              </div>
   700→              {result.errors.length > 0 && (
   701→                <ul className="mt-2 max-h-32 overflow-y-auto text-xs space-y-0.5">
   702→                  {result.errors.slice(0, 20).map((e, i) => (
   703→                    <li key={i}>• {e}</li>
   704→                  ))}
   705→                  {result.errors.length > 20 && <li>...and {result.errors.length - 20} more</li>}
   706→                </ul>
   707→              )}
   708→            </div>
   709→          )}
   710→        </CardContent>
   711→      </Card>
   712→    </div>
   713→  )
   714→}
