    43→    43→function tierFor(u: MapUtility): { label: string; color: string; ring: string } {
    44→    44→  if (u.legalExceedances > 0) {
    45→    45→    return { label: 'Above legal limit', color: '#e11d48', ring: '#fecdd3' }
    46→    46→  }
    47→    47→  if (u.healthExceedances >= 5) {
    48→    48→    return { label: 'Many health exceedances', color: '#f59e0b', ring: '#fde68a' }
    49→    49→  }
    50→    50→  if (u.healthExceedances > 0) {
    51→    51→    return { label: 'Some health exceedances', color: '#06b6d4', ring: '#a5f3fc' }
    52→    52→  }
    53→    53→  return { label: 'Within guidelines', color: '#10b981', ring: '#a7f3d0' }
    54→    54→}
